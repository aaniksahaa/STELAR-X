#   NORMALIZED_TRIPLET_SCORE: <value>

# ============================================================================
# Technical details:
# ============================================================================
#
# TRIPLET scoring formula (STELAR-X):
#   score = p1 * p2 * (p1 + p2 - 2) / 2
#   where p1, p2 are intersection sizes
#
# QUARTET scoring formula (ASTRAL):
#   F(a,b,c) = a * b * c * (a + b + c - 3) / 2
#   Uses 3x3 intersection grid with 6-term QI computation
#   Optimized: only 4 intersections computed, rest derived from row/column sums
