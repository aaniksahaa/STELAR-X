# Basic usage (uses default output files)
./treeFromTriplets -fit example1.tree

# Specify output tree file with names
./treeFromTriplets -fit example1.tree -frt output.tree

# Specify both output tree files (with names and with numbers)
./treeFromTriplets -fit example1.tree -frt output.tree -frtN output_numbers.tree

# Specify output tree file and log file
./treeFromTriplets -fit example1.tree -frt output.tree -flg output.log

# Full example with all output options
./treeFromTriplets -fit example1.tree -frt output.tree -frtN output_numbers.tree -flg output.log

# With different input files
./treeFromTriplets -fit example2.tree -frt result2.tree
./treeFromTriplets -fit example3.tree -frt result3.tree -flg result3.log

# Limit number of triplets with -nd
./treeFromTriplets -fit example1.tree -frt output.tree -nd 1000
